# Browser Visuals

Run through `npm run test:browser -- visuals`. Uses the local generated account from the ignored .env. This scenario captures the seven primary screens and checks overflow on all seven routes at three additional viewport sizes. It does not create business records. HEADLESS=1 is available for CI.

```javascript
const { chromium } = require(process.cwd()+'/node_modules/playwright');
const fs=require('fs');
const ROOT=process.cwd();
const env=require(process.cwd()+'/node_modules/dotenv').config({path:ROOT+'/.env',quiet:true}).parsed||{};
const TARGET_URL=process.env.TARGET_URL||'http://localhost:5177';
(async()=>{const browser=await chromium.launch({headless:process.env.HEADLESS==='1'});const page=await browser.newPage({viewport:{width:1536,height:1024}});const errors=[];page.on('pageerror',e=>errors.push(e.message));page.on('response',r=>{if(r.url().includes('/api/')&&r.status()>=400&&r.status()!==401)errors.push(r.status()+' '+r.url());});await page.goto(TARGET_URL+'/login');await page.getByLabel('Email Address').fill(env.ADMIN_EMAIL);await page.getByLabel('Password',{exact:true}).fill(env.ADMIN_PASSWORD);await page.getByRole('button',{name:'Sign In',exact:true}).click();await page.waitForURL('**/dashboard',{timeout:10000}).catch(async e=>{console.log(await page.locator('body').innerText());await page.screenshot({path:ROOT+'/docs/verification/login-error.png'});throw e;});await page.getByRole('heading',{name:/Good (Morning|Afternoon|Evening), Parvath!/}).waitFor();await page.screenshot({path:ROOT+'/docs/verification/01-dashboard.png',fullPage:false});let profile='';for(const [slug,num] of [['clients','02-clients'],['clients/new','04-add-client'],['leads','05-leads'],['renewals','06-renewals'],['followups','07-followups']]){await page.goto(TARGET_URL+'/'+slug);await page.waitForLoadState('networkidle');if(slug==='clients'){const r=await page.request.get(TARGET_URL+'/api/clients?q=Rajesh');profile=(await r.json()).data[0].id;}await page.screenshot({path:ROOT+'/docs/verification/'+num+'.png',fullPage:false});}await page.goto(TARGET_URL+'/clients/'+profile);await page.waitForLoadState('networkidle');await page.screenshot({path:ROOT+'/docs/verification/03-client-profile.png',fullPage:false});for(const [width,height,name] of [[1280,900,'desktop'],[768,1024,'tablet'],[390,844,'mobile']]){await page.setViewportSize({width,height});await page.goto(TARGET_URL+'/dashboard');await page.waitForLoadState('networkidle');await page.screenshot({path:ROOT+'/docs/verification/'+name+'-dashboard.png',fullPage:true});if(await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth))throw new Error(name+' dashboard overflow');console.log(name+' dashboard fits');if(width<1000){await page.getByRole('button',{name:'Open navigation',exact:true}).click();for(let i=0;i<14;i++){await page.keyboard.press('Tab');if(!await page.evaluate(()=>document.querySelector('.sidebar').contains(document.activeElement)))throw new Error(name+' drawer focus escaped');}await page.keyboard.press('Escape');if(!await page.getByRole('button',{name:'Open navigation',exact:true}).evaluate(el=>el===document.activeElement))throw new Error(name+' drawer did not restore focus');await page.getByRole('button',{name:'Open navigation',exact:true}).click();await page.getByRole('navigation').getByRole('link',{name:'Clients',exact:true}).click();await page.waitForURL('**/clients');console.log(name+' drawer navigation passed');}await page.goto(TARGET_URL+'/clients/new');await page.waitForLoadState('networkidle');await page.screenshot({path:ROOT+'/docs/verification/'+name+'-onboarding.png',fullPage:true});if(await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth))throw new Error(name+' onboarding overflow');console.log(name+' onboarding fits');for(const route of ['clients','clients/'+profile,'leads','renewals','followups']){await page.goto(TARGET_URL+'/'+route);await page.waitForLoadState('networkidle');if(await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth))throw new Error(name+' '+route+' overflow');}console.log(name+' all seven routes fit');}console.log('BROWSER_ERRORS',JSON.stringify(errors));await browser.close();if(errors.length)throw new Error('Browser errors detected');})().catch(e=>{console.error(e);process.exit(1)});

```
